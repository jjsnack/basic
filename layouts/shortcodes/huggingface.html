{{- /* {{< huggingface "owner/model" >}} — card linking to a Hugging Face model. Pass type="dataset" for a dataset repo. Auto-fetches pipeline tag/likes/downloads at build time; pass desc="…" likes="…" downloads="…" to override. */ -}} {{- $repo := .Get 0 -}} {{- with .Get "repo" }}{{ $repo = . }}{{ end -}} {{- if not $repo }}{{ errorf "huggingface shortcode: repo required, e.g. {{}} (%s)" .Position }}{{ end -}} {{- $kind := .Get "type" | default "model" -}} {{- $endpoint := "models" -}} {{- if eq $kind "dataset" }}{{ $endpoint = "datasets" }}{{ end -}} {{- $desc := .Get "desc" -}} {{- $pipeline := "" -}} {{- $likes := .Get "likes" -}} {{- $downloads := .Get "downloads" -}} {{- if not (and $likes $downloads) -}} {{- $try := try (resources.GetRemote (printf "https://huggingface.co/api/%s/%s" $endpoint $repo)) -}} {{- with $try -}} {{- if not .Err -}} {{- with .Value -}} {{- $data := transform.Unmarshal .Content -}} {{- $pipeline = $data.pipeline_tag -}} {{- if not $likes }}{{ $likes = int $data.likes }}{{ end -}} {{- if not $downloads }}{{ $downloads = int $data.downloads }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} 🤗 {{ $repo }} {{- with $desc }}{{ . }}{{ end }} {{- if or $pipeline $likes $downloads }} {{- with $pipeline }}{{ . }}{{ end }} {{- with $downloads }}⬇ {{ . }}{{ end }} {{- with $likes }}♥ {{ . }}{{ end }} {{- end }}
