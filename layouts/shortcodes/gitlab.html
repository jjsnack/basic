{{- /* {{< gitlab "namespace/project" >}} — card linking to a GitLab repo. Auto-fetches description/stars/forks from the GitLab API at build time; pass desc="…" stars="…" forks="…" to override (or skip the network call). */ -}} {{- $repo := .Get 0 -}} {{- with .Get "repo" }}{{ $repo = . }}{{ end -}} {{- if not $repo }}{{ errorf "gitlab shortcode: repo required, e.g. {{}} (%s)" .Position }}{{ end -}} {{- $desc := .Get "desc" -}} {{- $stars := .Get "stars" -}} {{- $forks := .Get "forks" -}} {{- if not (and $desc $stars $forks) -}} {{- $id := replace $repo "/" "%2F" -}} {{- $try := try (resources.GetRemote (printf "https://gitlab.com/api/v4/projects/%s" $id)) -}} {{- with $try -}} {{- if not .Err -}} {{- with .Value -}} {{- $data := transform.Unmarshal .Content -}} {{- if not $desc }}{{ $desc = $data.description }}{{ end -}} {{- if not $stars }}{{ $stars = int $data.star_count }}{{ end -}} {{- if not $forks }}{{ $forks = int $data.forks_count }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{ $repo }} {{- with $desc }}{{ . }}{{ end }} {{- if or $stars $forks }} {{- with $stars }}★ {{ . }}{{ end }} {{- with $forks }}⑂ {{ . }}{{ end }} {{- end }}
