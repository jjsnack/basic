{{- /* {{< github "owner/repo" >}} — card linking to a GitHub repo. Auto-fetches description/language/stars from the GitHub API at build time; pass desc="…" lang="…" stars="…" to override (or skip the network call). */ -}} {{- $repo := .Get 0 -}} {{- with .Get "repo" }}{{ $repo = . }}{{ end -}} {{- if not $repo }}{{ errorf "github shortcode: repo required, e.g. {{}} (%s)" .Position }}{{ end -}} {{- $desc := .Get "desc" -}} {{- $lang := .Get "lang" -}} {{- $stars := .Get "stars" -}} {{- if not (and $desc $lang $stars) -}} {{- $try := try (resources.GetRemote (printf "https://api.github.com/repos/%s" $repo)) -}} {{- with $try -}} {{- if not .Err -}} {{- with .Value -}} {{- $data := transform.Unmarshal .Content -}} {{- if not $desc }}{{ $desc = $data.description }}{{ end -}} {{- if not $lang }}{{ $lang = $data.language }}{{ end -}} {{- if not $stars }}{{ $stars = int $data.stargazers_count }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{ $repo }} {{- with $desc }}{{ . }}{{ end }} {{- if or $lang $stars }} {{- with $lang }}{{ . }}{{ end }} {{- with $stars }}★ {{ . }}{{ end }} {{- end }}
