{{- /* {{< npm "left-pad" >}} — card linking to an npm package. Auto-fetches description/version/license from the registry at build time; pass desc="…" version="…" license="…" to override (or skip the network call). */ -}} {{- $pkg := .Get 0 -}} {{- with .Get "pkg" }}{{ $pkg = . }}{{ end -}} {{- if not $pkg }}{{ errorf "npm shortcode: package name required, e.g. {{}} (%s)" .Position }}{{ end -}} {{- $desc := .Get "desc" -}} {{- $version := .Get "version" -}} {{- $license := .Get "license" -}} {{- if not (and $desc $version $license) -}} {{- $try := try (resources.GetRemote (printf "https://registry.npmjs.org/%s" $pkg)) -}} {{- with $try -}} {{- if not .Err -}} {{- with .Value -}} {{- $data := transform.Unmarshal .Content -}} {{- if not $desc }}{{ $desc = $data.description }}{{ end -}} {{- if not $version }}{{ $version = index $data "dist-tags" "latest" }}{{ end -}} {{- if not $license }}{{ $license = $data.license }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} 📦 {{ $pkg }} {{- with $desc }}{{ . }}{{ end }} {{- if or $version $license }} {{- with $version }}v{{ . }}{{ end }} {{- with $license }}{{ . }}{{ end }} {{- end }}
